let scene, camera, renderer, controls;
let loader = new THREE.GLTFLoader();
let models = [];

// === VARIABILI GLOBALI ===
let keys = {};
let velocity = new THREE.Vector3();
let direction = new THREE.Vector3();
let walkTime = 0;
let stepTime = 0;
let baseCameraHeight = 1.6;

let stepSounds = [];
let listener, audioLoader;

// MODEL DATA FROM DJANGO
let modelsData = window.MODELS_FROM_DJANGO || [];

// =========================== INIT ===========================
function init() {
    // === SCENA ===
    scene = new THREE.Scene();
    scene.background = new THREE.Color(0xcccccc);

    // === LUCI ===
    scene.add(new THREE.AmbientLight(0xffffff, 0.8));

    // Direzionale (sole) con ombre
    const sunLight = new THREE.DirectionalLight(0xffffff, 1);
    sunLight.position.set(10, 20, 10);
    sunLight.castShadow = true;
    sunLight.shadow.mapSize.width = 2048;
    sunLight.shadow.mapSize.height = 2048;
    sunLight.shadow.camera.near = 0.5;
    sunLight.shadow.camera.far = 50;
    sunLight.shadow.camera.left = -20;
    sunLight.shadow.camera.right = 20;
    sunLight.shadow.camera.top = 20;
    sunLight.shadow.camera.bottom = -20;
    scene.add(sunLight);

    // === RENDERER ===
    const canvas = document.getElementById("room-canvas");
    renderer = new THREE.WebGLRenderer({ canvas: canvas });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;

    // === CAMERA ===
    camera = new THREE.PerspectiveCamera(
        65, window.innerWidth / window.innerHeight, 0.1, 1000
    );
    camera.position.set(0, 1.6, 5);

    // === FOOTSTEP ===
    setupFootsteps();

    // === CONTROLLI POINTER LOCK ===
    controls = new THREE.PointerLockControls(camera, document.body);
    document.body.addEventListener("click", () => controls.lock());

    document.addEventListener("keydown", e => keys[e.code] = true);
    document.addEventListener("keyup", e => keys[e.code] = false);

    // === STANZA 3D ===
    buildRoom();

    // === CARICAMENTO MODELLI ===
    loadUserModels();

    // === LOOP ===
    movePlayer();
    animate();

    window.addEventListener("resize", onResize);

    // === LOADING SCREEN ===
    window.addEventListener("DOMContentLoaded", () => {
        setTimeout(hideLoading, 500); // 0.5s prima fade
    });
}

// =========================== FOOTSTEP ===========================
function setupFootsteps() {
    listener = new THREE.AudioListener();
    camera.add(listener);
    audioLoader = new THREE.AudioLoader();

    const stepFiles = [
        "/static/sounds/Step1.mp3","/static/sounds/Step2.mp3","/static/sounds/Step3.mp3",
        "/static/sounds/Step4.mp3","/static/sounds/Step5.mp3","/static/sounds/Step6.mp3",
        "/static/sounds/Step7.mp3","/static/sounds/Step8.mp3"
    ];

    stepFiles.forEach(file => {
        let sound = new THREE.Audio(listener);
        audioLoader.load(file, buffer => {
            sound.setBuffer(buffer);
            sound.setVolume(0.5);
        });
        stepSounds.push(sound);
    });
}

function playStepSound() {
    if (stepSounds.length === 0) return;
    let snd = stepSounds[Math.floor(Math.random() * stepSounds.length)];
    if (!snd.isPlaying) snd.play();
}

// =========================== MOVIMENTO ===========================
function movePlayer() {
    const speed = 0.1;
    direction.set(0, 0, 0);

    if (keys["KeyW"] || keys["ArrowUp"]) direction.z += 1;
    if (keys["KeyS"] || keys["ArrowDown"]) direction.z -= 1;
    if (keys["KeyA"] || keys["ArrowLeft"]) direction.x -= 1;
    if (keys["KeyD"] || keys["ArrowRight"]) direction.x += 1;

    let isMoving = direction.x !== 0 || direction.z !== 0;

    if (isMoving) {
        stepTime++;
        if (stepTime >= 25) {
            playStepSound();
            stepTime = 0;
        }
    } else {
        stepTime = 20;
    }

    direction.normalize();
    velocity.copy(direction).multiplyScalar(speed);

    controls.moveRight(velocity.x);
    controls.moveForward(velocity.z);

    requestAnimationFrame(movePlayer);
}

// =========================== STANZA ===========================
let floor;
function buildRoom() {
    floor = new THREE.Mesh(
        new THREE.PlaneGeometry(20, 20),
        new THREE.MeshStandardMaterial({ color: 0x808080 })
    );
    floor.rotation.x = -Math.PI / 2;
    floor.receiveShadow = true;
    scene.add(floor);

    function wall(x, z, rot = 0) {
        const w = new THREE.Mesh(
            new THREE.BoxGeometry(20, 5, 0.2),
            new THREE.MeshStandardMaterial({ color: 0xaaaaaa })
        );
        w.position.set(x, 2.5, z);
        w.rotation.y = rot;
        w.castShadow = true;
        w.receiveShadow = true;
        scene.add(w);
    }
    wall(0, -10);
    wall(0, 10);
    wall(-10, 0, Math.PI / 2);
    wall(10, 0, Math.PI / 2);
}

// =========================== MODELLI ===========================
function loadUserModels() {
    let remaining = modelsData.length;
    if (remaining === 0) hideLoading();

    modelsData.forEach(m => {
        loader.load(m.url, gltf => {
            const model = gltf.scene;
            model.scale.set(m.scale, m.scale, m.scale);
            model.position.set(m.pos[0], m.pos[1], m.pos[2]);
            model.traverse(child => {
                if (child.isMesh) {
                    child.castShadow = true;
                    child.receiveShadow = true;
                }
            });
            scene.add(model);

            if (--remaining === 0) hideLoading();
        });
    });
}

// =========================== LOADING SCREEN ===========================
function hideLoading() {
    const screen = document.getElementById("loading-screen");
    if (!screen) return;
    screen.classList.add("hidden");
    setTimeout(() => {
        screen.style.display = "none";
    }, 1000);
}

// =========================== ANIMATE ===========================
function animate() {
    requestAnimationFrame(animate);

    let isMoving =
        keys["KeyW"] || keys["KeyA"] || keys["KeyS"] || keys["KeyD"] ||
        keys["ArrowUp"] || keys["ArrowLeft"] || keys["ArrowDown"] || keys["ArrowRight"];

    if (isMoving) {
        walkTime += 0.25;
        camera.position.y = baseCameraHeight + Math.sin(walkTime) * 0.08;
    } else {
        camera.position.y += (baseCameraHeight - camera.position.y) * 0.1;
    }

    renderer.render(scene, camera);
}

// =========================== RESIZE ===========================
function onResize() {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
}

// =========================== START ===========================
window.addEventListener("DOMContentLoaded", init);
